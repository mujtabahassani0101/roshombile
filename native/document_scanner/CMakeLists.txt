cmake_minimum_required(VERSION 3.22.1)
project(yse_document_scanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  add_library(document_scanner_opencv SHARED IMPORTED)
  set_target_properties(document_scanner_opencv PROPERTIES
    IMPORTED_LOCATION "${OMR_OPENCV_SHARED_DIR}/${ANDROID_ABI}/libopencv_java4.so"
    INTERFACE_INCLUDE_DIRECTORIES "${OMR_OPENCV_INCLUDE_DIR}")
  set(DOCUMENT_OPENCV document_scanner_opencv)
else()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
  set(DOCUMENT_OPENCV ${OpenCV_LIBS})
endif()
add_library(document_scanner_core STATIC src/DocumentScanner.cpp)
target_include_directories(document_scanner_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(document_scanner_core PUBLIC ${DOCUMENT_OPENCV})
target_compile_options(document_scanner_core PRIVATE -O2)
if(ANDROID)
  add_library(document_scanner SHARED src/ScannerJni.cpp)
  target_link_libraries(document_scanner PRIVATE document_scanner_core)
else()
  add_executable(document-scanner src/main.cpp)
  target_link_libraries(document-scanner PRIVATE document_scanner_core)
endif()
